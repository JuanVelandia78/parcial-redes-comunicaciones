# 11 - NAT y reenvio en el kernel del host

$ cat /proc/sys/net/ipv4/ip_forward
1

$ iptables -t nat -S  (host)
-N DOCKER
-A PREROUTING -m addrtype --dst-type LOCAL -j DOCKER
-A OUTPUT ! -d 127.0.0.0/8 -m addrtype --dst-type LOCAL -j DOCKER
-A POSTROUTING -s 172.20.0.0/24 ! -o br-frontend -j MASQUERADE
-A POSTROUTING -s 172.17.0.0/16 ! -o docker0 -j MASQUERADE
-A DOCKER ! -i br-frontend -p tcp -m tcp --dport 80 -j DNAT --to-destination 172.20.0.5:80

$ iptables -S DOCKER-ISOLATION / internal (host)
-A DOCKER -d 172.20.0.5/32 ! -i br-frontend -o br-frontend -p tcp -m tcp --dport 80 -j ACCEPT
-A DOCKER ! -i br-frontend -o br-frontend -j DROP
-A DOCKER-BRIDGE -o br-frontend -j DOCKER
-A DOCKER-CT -o br-frontend -m conntrack --ctstate RELATED,ESTABLISHED -j ACCEPT
-A DOCKER-FORWARD -i br-backend -o br-backend -j ACCEPT
-A DOCKER-FORWARD -i br-frontend -j ACCEPT
-A DOCKER-INTERNAL ! -s 172.21.0.0/24 -o br-backend -j DROP
-A DOCKER-INTERNAL ! -d 172.21.0.0/24 -i br-backend -j DROP
